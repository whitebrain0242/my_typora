```go
func maxSubArray(nums []int) int {
    ans:=math.MinInt
    sum:=0
    minS:=0
    for _,num:=range nums{
        sum+=num
        ans=max(ans,sum-minS)
        if sum<minS{
            minS=sum
        }
    }
    return ans 
}

```
