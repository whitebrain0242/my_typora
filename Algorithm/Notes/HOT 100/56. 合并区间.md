```go
func merge(intervals [][]int) [][]int {
    slices.SortFunc(intervals,func(p,q []int)int{return p[0]-q[0]})
    ans:=[][]int{}
    for _,p:=range intervals{
        if len(ans)>0&&p[0]<=ans[len(ans)-1][1]{
            ans[len(ans)-1][1]=max(p[1],ans[len(ans)-1][1])
        }else{
            ans=append(ans,p)
        }
    }
    return ans 
}
```
