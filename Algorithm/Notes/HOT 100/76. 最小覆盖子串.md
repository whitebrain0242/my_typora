```go
func cover(cntS ,cntT []int)bool{
    for i:='A';i<='Z';i++{
        if cntS[i]<cntT[i]{
            return false
        }
    }
    for j:='a';j<='z';j++{
        if cntS[j]<cntT[j]{
            return false
        }
    }
    return true
}
func minWindow(s string, t string) string {
    var cntS,cntT [128]int
    //先把字串有哪字符存进去
    for _,c:=range t{
        cntT[c]++
    }

    ansL,ansR:=-1,len(s)-1//最初的字串长度
    left:=0
    for i,c:=range s{
        cntS[c]++
        //先看能不能涵盖
        for cover(cntS[:],cntT[:]){
            //更新当前的字串下标,然后开始一个一个删除左元素
            if i-left<ansR-ansL{
                ansL,ansR=left,i
            }
            cntS[s[left]]--
            left++
        }
        
    }
    if ansL<0{
        return ""
    }
    return s[ansL:ansR+1]
}
```
