```go
func threeSum(nums []int) [][]int {
    slices.Sort(nums)
    ans :=[][]int{}
    
    for i ,num:=range nums[:len(nums)-2]{
        if i>0&&nums[i]==nums[i-1]{
            continue
        }
        //有两个优化：如果这个数和后面最大的加起来还是小或者是和后面最小的加起来还是大的话就不符合
        if num+nums[i+1]+nums[i+2]>0{
            break
        }
        if num+nums[len(nums)-1]+nums[len(nums)-2]<0{
            continue
        }
        p1,p2:=i+1,len(nums)-1
        for p1<p2{
            s:=num+nums[p1]+nums[p2]
            if s>0{
                p2--
            }else if s<0{
                p1++
            }else{
                ans=append(ans,[]int{num,nums[p1],nums[p2]})
                for p1++;p1<p2&&nums[p1]==nums[p1-1];p1++{}
                for p2--;p2>p1&&nums[p2]==nums[p2+1];p2--{}
            }

        }
    }
    return ans
    
}
```


## 参考思路

（独立完成后再整理）
1. 就是遍历每一个数字，然后使用双指针进行判断就好了
2. 重点其实是优化：当前数字与最大的两个数相加还是太小的话就直接下一个数字
   还有一个优化：当前数字与后面相邻的最小的两个数相加还是太大的话就直接跳出循环
3. 还有一个重点就是不去遍历重复数字，因为可能会造成输出重复

## 时间复杂度
