类似 Linux：

```bash
ls -l
```

例如：

```text

```

## NLST

类似：

```bash
ls
```

只发送名字：

```text
docs
```

# 辅助函数

## sendAll()

确保全部发送完成。

```cpp
bool sendAll(
    int fd,
    const std::string& data)
{
    size_t sent = 0;

    while (sent < data.size())
    {
        ssize_t n =
            send(
                fd,
                data.data() + sent,
                data.size() - sent,
                0);

        if (n <= 0)
        {
            return false;
        }

        sent += n;
    }

    return true;
}
```

# formatListLine()

生成 LIST 输出。

```cpp
std::string formatListLine(
    const fs::directory_entry& entry)
{
    std::string perms;

    if (entry.is_directory())
        perms = "drwxr-xr-x";
    else
        perms = "-rw-r--r--";

    uintmax_t size = 0;

    if (entry.is_regular_file())
    {
        size = entry.file_size();
    }

    std::ostringstream oss;

    oss
        << perms
        << " 1 owner group "
        << size
        << " "
        << entry.path().filename().string()
        << "\r\n";

    return oss.str();
}
```

# handleLIST()

新增：

```cpp
void handleLIST()
{
    if (!loggedIn)
    {
        sendReply(
            530,
            "Please login");
        return;
    }

    if (pasvListenFd < 0)
    {
        sendReply(
            425,
            "Use PASV first");

        return;
    }

    sendReply(
        150,
        "Opening ASCII mode data connection");

    int dataFd =
        acceptDataConnection();

    if (dataFd < 0)
    {
        sendReply(
            425,
            "Data connection failed");

        return;
    }

    try
    {
        for (const auto& entry :
             fs::directory_iterator(
                 currentDir))
        {
            auto line =
                formatListLine(entry);

            sendAll(
                dataFd,
                line);
        }

        close(dataFd);

        sendReply(
            226,
            "Transfer complete");
    }
    catch (...)
    {
        close(dataFd);

        sendReply(
            550,
            "Directory read failed");
    }
}
```

# handleNLST()

新增：

```cpp
void handleNLST()
{
    if (!loggedIn)
    {
        sendReply(
            530,
            "Please login");

        return;
    }

    if (pasvListenFd < 0)
    {
        sendReply(
            425,
            "Use PASV first");

        return;
    }

    sendReply(
        150,
        "Opening data connection");

    int dataFd =
        acceptDataConnection();

    if (dataFd < 0)
    {
        sendReply(
            425,
            "Data connection failed");

        return;
    }

    try
    {
        for (const auto& entry :
             fs::directory_iterator(
                 currentDir))
        {
            std::string line =
                entry.path()
                    .filename()
                    .string();

            line += "\r\n";

            sendAll(
                dataFd,
                line);
        }

        close(dataFd);

        sendReply(
            226,
            "Transfer complete");
    }
    catch (...)
    {
        close(dataFd);

        sendReply(
            550,
            "Directory read failed");
    }
}
```
