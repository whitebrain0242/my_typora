# ftp_server_stage1.cpp

```cpp
// ftp_server_stage1.cpp
//
// Stage 1：最小 TCP 服务器
//
// 功能：
//   1. 监听 2100 端口
//   2. 接受客户端连接
//   3. 发送 FTP 欢迎消息
//   4. 接收客户端输入
//   5. 打印客户端发送的内容
//
// 编译：
//   g++ -std=c++17 ftp_server_stage1.cpp -o ftp_server
//
// 运行：
//   ./ftp_server
//
// 测试：
//   nc 127.0.0.1 2100
//
// 输入：
//   hello
//   USER test
//
// 服务器会打印收到的数据
//

#include <arpa/inet.h>
#include <cstring>
#include <iostream>
#include <netinet/in.h>
#include <sys/socket.h>
#include <unistd.h>

constexpr int FTP_PORT = 2100;
constexpr int BUFFER_SIZE = 4096;

int main()
{
    //----------------------------------------------------------
    // 1. 创建 TCP Socket
    //----------------------------------------------------------

    int serverFd = socket(
        AF_INET,      // IPv4
        SOCK_STREAM,  // TCP
        0
    );

    if (serverFd < 0)
    {
        perror("socket");
        return 1;
    }

    std::cout << "[INFO] socket created\n";

    //----------------------------------------------------------
    // 2. SO_REUSEADDR
    //----------------------------------------------------------

    int opt = 1;

    if (setsockopt(
            serverFd,
            SOL_SOCKET,
            SO_REUSEADDR,
            &opt,
            sizeof(opt)) < 0)
    {
        perror("setsockopt");
        close(serverFd);
        return 1;
    }

    //----------------------------------------------------------
    // 3. 配置监听地址
    //----------------------------------------------------------

    sockaddr_in serverAddr {};

    serverAddr.sin_family = AF_INET;

    // 监听所有网卡
    serverAddr.sin_addr.s_addr = htonl(INADDR_ANY);

    serverAddr.sin_port = htons(FTP_PORT);

    //----------------------------------------------------------
    // 4. bind
    //----------------------------------------------------------

    if (bind(
            serverFd,
            reinterpret_cast<sockaddr*>(&serverAddr),
            sizeof(serverAddr)) < 0)
    {
        perror("bind");
        close(serverFd);
        return 1;
    }

    std::cout << "[INFO] bind success\n";

    //----------------------------------------------------------
    // 5. listen
    //----------------------------------------------------------

    if (listen(serverFd, 5) < 0)
    {
        perror("listen");
        close(serverFd);
        return 1;
    }

    std::cout
        << "[INFO] listening on port "
        << FTP_PORT
        << "\n";

    //----------------------------------------------------------
    // 6. 等待客户端
    //----------------------------------------------------------

    while (true)
    {
        sockaddr_in clientAddr {};
        socklen_t clientLen = sizeof(clientAddr);

        int clientFd = accept(
            serverFd,
            reinterpret_cast<sockaddr*>(&clientAddr),
            &clientLen);

        if (clientFd < 0)
        {
            perror("accept");
            continue;
        }

        //------------------------------------------------------
        // 显示客户端地址
        //------------------------------------------------------

        char ip[INET_ADDRSTRLEN];

        inet_ntop(
            AF_INET,
            &clientAddr.sin_addr,
            ip,
            sizeof(ip));

        std::cout
            << "\n[INFO] client connected: "
            << ip
            << ":"
            << ntohs(clientAddr.sin_port)
            << "\n";

        //------------------------------------------------------
        // 7. FTP 欢迎消息
        //------------------------------------------------------

        const char* welcome =
            "220 Simple FTP Server Ready\r\n";

        send(
            clientFd,
            welcome,
            strlen(welcome),
            0);

        //------------------------------------------------------
        // 8. 接收客户端数据
        //------------------------------------------------------

        char buffer[BUFFER_SIZE];

        while (true)
        {
            memset(buffer, 0, sizeof(buffer));

            ssize_t bytes = recv(
                clientFd,
                buffer,
                sizeof(buffer) - 1,
                0);

            //--------------------------------------------------
            // 客户端关闭连接
            //--------------------------------------------------

            if (bytes == 0)
            {
                std::cout
                    << "[INFO] client disconnected\n";

                break;
            }

            //--------------------------------------------------
            // 出错
            //--------------------------------------------------

            if (bytes < 0)
            {
                perror("recv");
                break;
            }

            //--------------------------------------------------
            // 打印收到的数据
            //--------------------------------------------------

            std::cout
                << "[CLIENT] "
                << buffer;

            //--------------------------------------------------
            // 回显给客户端
            //--------------------------------------------------

            std::string reply =
                "200 OK: " +
                std::string(buffer);

            send(
                clientFd,
                reply.c_str(),
                reply.size(),
                0);
        }

        //------------------------------------------------------
        // 关闭客户端连接
        //------------------------------------------------------

        close(clientFd);
    }

    close(serverFd);

    return 0;
}
```

这一版最重要的是理解整个 TCP 服务器生命周期：

```text
socket()
    ↓
setsockopt()
    ↓
bind()
    ↓
listen()
    ↓
accept()
    ↓
send()
recv()
    ↓
close()
```
