很好，到了 **Stage4：PASV 数据连接**。

这是 FTP 最容易让人困惑的一部分，因为前面三版都只有一个 socket：

```text
客户端
   │
   ▼
控制连接(2100)
   │
   ▼
服务器
```

但是 FTP 真正工作时是：

```text
                控制连接
客户端  <---------------------->  2100

                数据连接
客户端  <---------------------->  50001
```

也就是说：

```text
命令走控制连接
数据走数据连接
```

例如：

```text
LIST
RETR
STOR
```

都必须额外建立一个数据连接。

# 新增创建监听Socket

```cpp
int createListenSocket(int port)
{
    int fd =
        socket(
            AF_INET,
            SOCK_STREAM,
            0);

    if (fd < 0)
    {
        return -1;
    }

    int opt = 1;

    setsockopt(
        fd,
        SOL_SOCKET,
        SO_REUSEADDR,
        &opt,
        sizeof(opt));

    sockaddr_in addr {};

    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr =
        htonl(INADDR_ANY);

    addr.sin_port =
        htons(port);

    if (bind(
            fd,
            reinterpret_cast<sockaddr*>(&addr),
            sizeof(addr)) < 0)
    {
        close(fd);
        return -1;
    }

    if (listen(fd, 1) < 0)
    {
        close(fd);
        return -1;
    }

    return fd;
}
```

# handlePASV()

这是本阶段核心。

```cpp
void handlePASV()
{
    if (!loggedIn)
    {
        sendReply(
            530,
            "Please login");
        return;
    }

    closePassive();

    //--------------------------------------------------
    // 为简单起见固定50000端口
    // Stage7再改随机端口
    //--------------------------------------------------

    pasvPort = 50000;

    pasvListenFd =
        createListenSocket(
            pasvPort);

    if (pasvListenFd < 0)
    {
        sendReply(
            425,
            "Cannot open passive connection");

        return;
    }

    //--------------------------------------------------
    // FTP格式:
    //
    // 127,0,0,1,195,80
    //
    // 195*256+80=50000
    //--------------------------------------------------

    sendReply(
        227,
        "Entering Passive Mode "
        "(127,0,0,1,195,80)");
}
```

------

# acceptDataConnection()

```cpp
int acceptDataConnection()
{
    if (pasvListenFd < 0)
    {
        return -1;
    }

    sockaddr_in addr {};

    socklen_t len =
        sizeof(addr);

    std::cout
        << "[PASV] waiting data connection...\n";

    int dataFd =
        accept(
            pasvListenFd,
            reinterpret_cast<sockaddr*>(
                &addr),
            &len);

    closePassive();

    if (dataFd >= 0)
    {
        std::cout
            << "[PASV] data connected\n";
    }

    return dataFd;
}
```

# handleTESTDATA()

```cpp
void handleTESTDATA()
{
    if (!loggedIn)
    {
        sendReply(
            530,
            "Please login");
        return;
    }

    sendReply(
        150,
        "Opening data connection");

    int dataFd =
        acceptDataConnection();

    if (dataFd < 0)
    {
        sendReply(
            425,
            "Data connection failed");

        return;
    }

    const char* msg =
        "Hello From Data Connection\r\n";

    send(
        dataFd,
        msg,
        strlen(msg),
        0);

    close(dataFd);

    sendReply(
        226,
        "Transfer complete");
}
```

