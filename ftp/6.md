void handleTYPE(
    const std::string& arg)
{
    if (arg == "I")
    {
        binaryMode = true;

        sendReply(
            200,
            "Binary mode");
    }
    else if (arg == "A")
    {
        binaryMode = false;

        sendReply(
            200,
            "ASCII mode");
    }
    else
    {
        sendReply(
            504,
            "Unknown TYPE");
    }
}

# sendFile()

新增辅助函数

```cpp
bool sendFile(
    int dataFd,
    const fs::path& file)
{
    constexpr size_t BUF_SIZE =
        64 * 1024;

    char buffer[BUF_SIZE];

    std::ifstream in(
        file,
        std::ios::binary);

    if (!in)
    {
        return false;
    }

    while (in)
    {
        in.read(
            buffer,
            sizeof(buffer));

        std::streamsize n =
            in.gcount();

        if (n <= 0)
        {
            break;
        }

        size_t sent = 0;

        while (sent <
               static_cast<size_t>(n))
        {
            ssize_t s =
                send(
                    dataFd,
                    buffer + sent,
                    n - sent,
                    0);

            if (s <= 0)
            {
                return false;
            }

            sent += s;
        }
    }

    return true;
}
```

# recvFile()

```cpp
bool recvFile(
    int dataFd,
    const fs::path& file)
{
    constexpr size_t BUF_SIZE =
        64 * 1024;

    char buffer[BUF_SIZE];

    std::ofstream out(
        file,
        std::ios::binary);

    if (!out)
    {
        return false;
    }

    while (true)
    {
        ssize_t n =
            recv(
                dataFd,
                buffer,
                sizeof(buffer),
                0);

        if (n == 0)
        {
            break;
        }

        if (n < 0)
        {
            return false;
        }

        out.write(
            buffer,
            n);
    }

    return true;
}
```

# handleRETR()

下载

```cpp
void handleRETR(
    const std::string& arg)
{
    if (!loggedIn)
    {
        sendReply(
            530,
            "Please login");
        return;
    }

    try
    {
        auto file =
            resolvePath(arg);

        if (!fs::exists(file))
        {
            sendReply(
                550,
                "File not found");

            return;
        }

        if (!fs::is_regular_file(file))
        {
            sendReply(
                550,
                "Not a file");

            return;
        }

        if (pasvListenFd < 0)
        {
            sendReply(
                425,
                "Use PASV first");

            return;
        }

        sendReply(
            150,
            "Opening data connection");

        int dataFd =
            acceptDataConnection();

        if (dataFd < 0)
        {
            sendReply(
                425,
                "Data connection failed");

            return;
        }

        bool ok =
            sendFile(
                dataFd,
                file);

        close(dataFd);

        if (ok)
        {
            sendReply(
                226,
                "Transfer complete");
        }
        else
        {
            sendReply(
                426,
                "Transfer aborted");
        }
    }
    catch (...)
    {
        sendReply(
            550,
            "Invalid path");
    }
}
```
