# ftp_server_stage2.cpp

```cpp
// ftp_server_stage2.cpp
//
// Stage2:
// FTP 控制命令框架
//
// 新增：
//   USER
//   PASS
//   QUIT
//   NOOP
//   SYST
//   PWD
//
// 编译：
// g++ -std=c++17 ftp_server_stage2.cpp -o ftp_server
//
// 测试：
// nc 127.0.0.1 2100
//

#include <arpa/inet.h>
#include <cstring>
#include <iostream>
#include <netinet/in.h>
#include <sstream>
#include <string>
#include <sys/socket.h>
#include <unistd.h>

constexpr int FTP_PORT = 2100;

class FtpSession
{
public:

    explicit FtpSession(int fd)
        : clientFd(fd)
    {
    }

    void run()
    {
        sendReply(220, "Simple FTP Server Ready");

        std::string line;

        while (recvLine(line))
        {
            std::string cmd;
            std::string arg;

            parseCommand(line, cmd, arg);

            std::cout
                << "[COMMAND] "
                << cmd
                << " "
                << arg
                << "\n";

            //--------------------------------------------------
            // USER
            //--------------------------------------------------

            if (cmd == "USER")
            {
                handleUSER(arg);
            }

            //--------------------------------------------------
            // PASS
            //--------------------------------------------------

            else if (cmd == "PASS")
            {
                handlePASS(arg);
            }

            //--------------------------------------------------
            // PWD
            //--------------------------------------------------

            else if (cmd == "PWD")
            {
                handlePWD();
            }

            //--------------------------------------------------
            // SYST
            //--------------------------------------------------

            else if (cmd == "SYST")
            {
                handleSYST();
            }

            //--------------------------------------------------
            // NOOP
            //--------------------------------------------------

            else if (cmd == "NOOP")
            {
                handleNOOP();
            }

            //--------------------------------------------------
            // QUIT
            //--------------------------------------------------

            else if (cmd == "QUIT")
            {
                handleQUIT();
                break;
            }

            //--------------------------------------------------
            // 未实现命令
            //--------------------------------------------------

            else
            {
                sendReply(
                    502,
                    "Command not implemented");
            }
        }
    }

private:

    int clientFd;

    bool loggedIn = false;

    std::string username;

    //----------------------------------------------------------
    // 发送 FTP 响应
    //----------------------------------------------------------

    void sendReply(
        int code,
        const std::string& text)
    {
        std::ostringstream oss;

        oss
            << code
            << " "
            << text
            << "\r\n";

        std::string msg = oss.str();

        send(
            clientFd,
            msg.c_str(),
            msg.size(),
            0);

        std::cout
            << "[REPLY] "
            << msg;
    }

    //----------------------------------------------------------
    // 接收一行
    //----------------------------------------------------------

    bool recvLine(std::string& line)
    {
        line.clear();

        char ch;

        while (true)
        {
            ssize_t n =
                recv(
                    clientFd,
                    &ch,
                    1,
                    0);

            if (n <= 0)
            {
                return false;
            }

            if (ch == '\n')
            {
                break;
            }

            if (ch != '\r')
            {
                line.push_back(ch);
            }
        }

        return true;
    }

    //----------------------------------------------------------
    // 拆命令
    //----------------------------------------------------------

    void parseCommand(
        const std::string& line,
        std::string& cmd,
        std::string& arg)
    {
        auto pos = line.find(' ');

        if (pos == std::string::npos)
        {
            cmd = line;
            arg.clear();
        }
        else
        {
            cmd = line.substr(0, pos);
            arg = line.substr(pos + 1);
        }

        for (char& c : cmd)
        {
            c = toupper(c);
        }
    }

    //----------------------------------------------------------
    // USER
    //----------------------------------------------------------

    void handleUSER(const std::string& arg)
    {
        username = arg;

        sendReply(
            331,
            "User name okay, need password");
    }

    //----------------------------------------------------------
    // PASS
    //----------------------------------------------------------

    void handlePASS(const std::string&)
    {
        loggedIn = true;

        sendReply(
            230,
            "Login successful");
    }

    //----------------------------------------------------------
    // PWD
    //----------------------------------------------------------

    void handlePWD()
    {
        if (!loggedIn)
        {
            sendReply(
                530,
                "Please login");
            return;
        }

        sendReply(
            257,
            "\"/\"");
    }

    //----------------------------------------------------------
    // SYST
    //----------------------------------------------------------

    void handleSYST()
    {
        sendReply(
            215,
            "UNIX Type: L8");
    }

    //----------------------------------------------------------
    // NOOP
    //----------------------------------------------------------

    void handleNOOP()
    {
        sendReply(
            200,
            "NOOP ok");
    }

    //----------------------------------------------------------
    // QUIT
    //----------------------------------------------------------

    void handleQUIT()
    {
        sendReply(
            221,
            "Goodbye");
    }
};

int main()
{
    //----------------------------------------------------------
    // socket
    //----------------------------------------------------------

    int serverFd =
        socket(
            AF_INET,
            SOCK_STREAM,
            0);

    if (serverFd < 0)
    {
        perror("socket");
        return 1;
    }

    //----------------------------------------------------------
    // reuseaddr
    //----------------------------------------------------------

    int opt = 1;

    setsockopt(
        serverFd,
        SOL_SOCKET,
        SO_REUSEADDR,
        &opt,
        sizeof(opt));

    //----------------------------------------------------------
    // bind
    //----------------------------------------------------------

    sockaddr_in addr {};

    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    addr.sin_port = htons(FTP_PORT);

    if (bind(
            serverFd,
            reinterpret_cast<sockaddr*>(&addr),
            sizeof(addr)) < 0)
    {
        perror("bind");
        return 1;
    }

    //----------------------------------------------------------
    // listen
    //----------------------------------------------------------

    if (listen(serverFd, 5) < 0)
    {
        perror("listen");
        return 1;
    }

    std::cout
        << "Listening on "
        << FTP_PORT
        << "\n";

    //----------------------------------------------------------
    // accept loop
    //----------------------------------------------------------

    while (true)
    {
        sockaddr_in clientAddr {};
        socklen_t len =
            sizeof(clientAddr);

        int clientFd =
            accept(
                serverFd,
                reinterpret_cast<sockaddr*>(
                    &clientAddr),
                &len);

        if (clientFd < 0)
        {
            continue;
        }

        char ip[INET_ADDRSTRLEN];

        inet_ntop(
            AF_INET,
            &clientAddr.sin_addr,
            ip,
            sizeof(ip));

        std::cout
            << "\nClient connected: "
            << ip
            << ":"
            << ntohs(clientAddr.sin_port)
            << "\n";

        FtpSession session(clientFd);

        session.run();

        close(clientFd);

        std::cout
            << "Client disconnected\n";
    }

    close(serverFd);

    return 0;
}
```
